# Save as Style (the owner's own Quick Edit styles): Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A project's actions sheet gains **Save as Style**: it keeps the project's music (own music copied), filter, transition, title (words + whole look) and photo motion as a STYLE on this phone. Quick Edit's sheet shows **Your styles** above the six built-in **Styles**; one of yours makes a draft exactly as a built-in does; touch and hold → Rename / Delete.

**Architecture:** `buildQuickEdit` stays the ONE builder: its input widens to `QuickRecipe | QuickPlan` and `music: AudioTrack | null`, and a recipe is mapped field for field by `planOfRecipe`, so the six built-ins build byte for byte as today (a PROOF snapshot committed first). A style is pure data (`src/projects/userStyles.ts`: `UserStyle`, `styleFromProject`, `planOfStyle`, `parseStyles`). It is stored by `makeStyleStore` (`src/projects/styleStore.ts`) in `<documents>/styles/styles.json` (+ `styles/media/` for own music), one serial chain, tmp + move, damaged files kept aside. The screens are Home's prompt, a `StyleTile` row in `QuickEditSheet`, and the system action sheet for Rename / Delete.

**Tech Stack:** Expo SDK 57, React Native (Fabric), TypeScript strict, zustand 5, Jest 29.7 (`jest-expo`) + RNTL 14. No native code, no new package, no `app.json` change, no schema bump.

**Spec:** `docs/superpowers/specs/2026-10-16-my-styles-design.md` (binding: §3 the rules of `styleFromProject`, §4 the builder and the PROOF, §5 music and beats, §6 storage, §7 the screens).

```ts
export function buildQuickEdit(a: { recipe: QuickRecipe | QuickPlan; project: Project; music: AudioTrack | null; beats: readonly number[]; titleId: string }): Project {
  const { project, music, beats, titleId } = a;
  const plan = "textTemplate" in a.recipe ? planOfRecipe(a.recipe) : a.recipe;
  if (project.clips.length === 0) return project;
  let p = project;
  const step = music ? markerStep(beats, plan.every) : 0;
  /** The clips that were given slack: id -> the source end of the target length WITHOUT it. */
  const exact = new Map<string, number>();
  if (step > 0) for (const c of project.clips) {
    if (isPhoto(c)) {
      p = trimClip(p, c.id, 0, r3(plan.hold * step + QUICK.slack));
      exact.set(c.id, r3(plan.hold * step));
    } else {
      const most = plan.videoHold * step;
      if (clipDuration(c) > most + QUICK.slack) {
        p = trimClip(p, c.id, c.trimStart, sourceAfter(c, c.trimStart, most + QUICK.slack));
        exact.set(c.id, sourceAfter(c, c.trimStart, most));
      }
    }
  }
  let total = totalDuration(p);
  if (music) {
    const first = beats.length > 0 ? Math.max(0, beats[0]) : 0;
    p = addAudioTrack(p, { ...music, start: 0, trimStart: first, trimEnd: music.sourceDuration, fadeIn: 0, fadeOut: 0 });
    const slacked = p.clips;

    // exact target length. Only that trailing run — nothing that was cut onto a beat is touched, and no cut just made can move.
    const same = (i: number) => p.clips[i].trimStart === slacked[i].trimStart && p.clips[i].trimEnd === slacked[i].trimEnd;
    let tail = slacked.length;
    while (tail > 0 && same(tail - 1)) tail--;
    for (const before of slacked.slice(tail)) {
      const end = exact.get(before.id);
      if (end !== undefined) p = trimClip(p, before.id, before.trimStart, end);
    }
    total = totalDuration(p);
    p = updateAudioTrackById(p, music.id, { trimEnd: Math.min(music.sourceDuration, first + total), fadeOut: QUICK.fadeOut });
  }
  if (p.beatMarkers.some((m) => m > total + 1e-9)) p = { ...p, beatMarkers: p.beatMarkers.filter((m) => m <= total + 1e-9) };
  if (plan.transition.type !== "none") for (const c of p.clips.slice(0, -1)) p = setTransition(p, c.id, plan.transition);
  if (plan.filter !== null) p = setFilterForAllClips(p, plan.filter, plan.filterIntensity);
  if (plan.motions.length > 0) {
    let n = 0;
    for (const c of p.clips) if (isPhoto(c)) p = setPhotoMotion(p, c.id, { id: plan.motions[n++ % plan.motions.length], strength: plan.motionStrength });
  }
  const end = Math.min(QUICK.titleSeconds, total);
  const t = plan.title;
  if (end > 0 && t) {
    if (t.from === "template") {
      p = addTextOverlay(p, makeOverlay({ id: titleId, text: t.text, x: 0.5, y: t.y, fontScale: t.fontScale, start: 0, end }));
      p = applyTextTemplate(p, titleId, t.templateId);
```

```ts
/** The most common key, a tie going to the one met first; null for none. */
function mostCommon<T>(keys: readonly T[]): T | null {
  const count = new Map<T, number>();
  for (const k of keys) count.set(k, (count.get(k) ?? 0) + 1);
  let best: T | null = null, most = 0;
  for (const [k, n] of count) if (n > most) { best = k; most = n; }   // Map keeps first-met order, so `>` keeps the first of a tie
  return best;
}

  const transition = type === null ? { type: "none" as const, duration: 0 } : { type, duration: cuts.find((t) => t.type === type)!.duration };
  let title: StyleTitle | null = null;
```

```ts
/** Marker steps for a style: a photo about `photoSeconds`, a video at most about `videoSeconds`, whatever the tempo. Without a step the values are unused (nothing is cut). */
export function paceFor(beats: readonly number[], every: 1 | 2 | 4): { hold: number; videoHold: number } {
  const step = markerStep(beats, every);
  if (step <= 0) return { hold: 1, videoHold: 1 };
  const hold = Math.min(USER_PACE.most, Math.max(1, Math.round(USER_PACE.photoSeconds / step)));
  return { hold, videoHold: Math.min(USER_PACE.most, Math.max(hold, Math.round(USER_PACE.videoSeconds / step))) };
}

/** What `buildQuickEdit` reads for a style, and the beats (FILE seconds) it cuts to. A bundled song: every 2nd shipped beat; own music: every one of its stored beats (the owner's markers). */

  const beats = s.music === null ? [] : s.music.kind === "own" ? s.music.beats : (BUNDLED_BEATS[s.music.trackId]?.beats ?? []);
  const every = s.music?.kind === "own" ? 1 : 2;

  return { beats, plan: { every, hold, videoHold, transition: s.transition, filter: s.filter, filterIntensity: s.filterIntensity,

const BLANK = makeOverlay({ id: "" });
function cleanMusic(v: unknown): StyleMusic | null {
  if (!isRec(v)) return null;
  if (v.kind === "bundled") return typeof v.trackId === "string" && v.trackId.length > 0 ? { kind: "bundled", trackId: v.trackId } : null;
  if (v.kind !== "own" || !isSafeName(v.file) || typeof v.title !== "string" || !isNum(v.durationSec) || v.durationSec <= 0) return null;

function cleanTitle(v: unknown): StyleTitle | null {
  if (!isRec(v) || typeof v.text !== "string" || v.text.trim().length === 0) return null;
  const l = isRec(v.look) ? v.look : {};
  const bg = isRec(l.background) && isHexColor(l.background.color) ? { color: l.background.color, opacity: num(l.background.opacity, 1, 0, 1) } : null;
  const look: TextLook = { fontId: (FONT_IDS as readonly unknown[]).includes(l.fontId) ? (l.fontId as FontId) : BLANK.fontId,
    fontScale: num(l.fontScale, BLANK.fontScale, OVERLAY_LIMITS.fontScale[0], OVERLAY_LIMITS.fontScale[1]), color: isHexColor(l.color) ? l.color : BLANK.color,
    background: bg, outline: typeof l.outline === "boolean" ? l.outline : BLANK.outline, style: clampTextStyle(l.style) };

  return { id: v.id, name: cut(v.name, STYLES.nameMax), createdAt: typeof v.createdAt === "string" ? v.createdAt : "", music: cleanMusic(v.music),
    filter, filterIntensity: num(v.filterIntensity, 1, 0, 1),
```

```ts
/** A built-in style by id, or one of the owner's with its own music's uri (asked of the store first; null when it has none). */
export type QuickChoice = QuickRecipeId | { style: UserStyle; musicUri: string | null };
/** Thrown before anything is created when a style's music cannot be had. */
export const STYLE_MUSIC_MISSING = "style music missing";

type Source = { plan: QuickRecipe | QuickPlan; music: { uri: string; title: string; durationSec: number } | null; beats: readonly number[] };
async function bundledSong(deps: QuickEditDeps, trackId: string, missing: () => Error) {
  const song = BUNDLED_TRACKS.find((t) => t.id === trackId);
  if (!song) throw missing();
  return { song, uri: await deps.assetUri(song.file) };
}
async function sourceOf(deps: QuickEditDeps, choice: QuickChoice): Promise<Source> {
  if (typeof choice === "string") {
    const recipe = QUICK_RECIPES[choice];
    const { song, uri } = await bundledSong(deps, recipe.trackId, () => new Error(`Quick edit: no bundled track "${recipe.trackId}"`));
    return { plan: recipe, music: { uri, title: song.title, durationSec: song.durationSec }, beats: BUNDLED_BEATS[song.id]?.beats ?? [] };
  }
  const { plan, beats } = planOfStyle(choice.style);
  const m = choice.style.music;
  if (m === null) return { plan, music: null, beats: [] };
  if (m.kind === "bundled") {
    const { song, uri } = await bundledSong(deps, m.trackId, () => new Error(STYLE_MUSIC_MISSING));
    return { plan, music: { uri, title: song.title, durationSec: song.durationSec }, beats };
  }
  if (!choice.musicUri) throw new Error(STYLE_MUSIC_MISSING);
  return { plan, music: { uri: choice.musicUri, title: m.title, durationSec: m.durationSec }, beats };
}

/**
 * Makes a Quick edit draft and returns its id: the style's music is resolved to a file, the picked media become a new project
 * (ratio Auto), the music is copied into it, `buildQuickEdit` builds the draft, and the draft is saved over the plain project.
 * All or nothing: the music is resolved FIRST (a bundled song is downloaded from the dev server in Expo Go), so when that fails
 * nothing is created or copied; when nothing could be imported no project is made (`createProject` throws and removes its folder);
 * when anything after that fails the new project is deleted again before the error is passed on. `failed` = picked items that
 * could not be read (the draft is made from the rest). A style without music makes a draft without sound.
 */
export async function makeQuickEdit(deps: QuickEditDeps, name: string, assets: PickedAsset[], choice: QuickChoice): Promise<{ id: string; failed: number }> {
  const source = await sourceOf(deps, choice);
  const { project, failed } = await deps.storage.createProject(name, assets);
  try {
    const music = source.music ? await deps.storage.importAudio(project.id, source.music, "music") : null;
    const draft = buildQuickEdit({ recipe: source.plan, project, music, beats: source.beats, titleId: deps.newId() });
    await deps.storage.saveProject(draft);
    return { id: project.id, failed };
  } catch (e) {
    await deps.storage.deleteProject(project.id).catch(() => {});
    throw e;
  }
}
```

```ts
import type { FsAdapter } from "./fs";
import { isSafeName } from "./projectFile";
import { STYLES, parseStyles, stylesText, type StyleDraft, type StyleMusic, type UserStyle } from "./userStyles";

// Where the owner's styles live (docs/superpowers/specs/2026-10-16-my-styles-design.md §6): <documents>/styles/styles.json and the
// own music in <documents>/styles/media/ — never under projects/, so no project sweep, list, pack or restore sees them. Every read
// and write is ONE turn of one chain, in the order asked (the phone's move is queued and removes the destination first).

export type StyleErrorCode = "full" | "newer" | "missing";
export class StyleStoreError extends Error {
  constructor(readonly code: StyleErrorCode) { super(`styles: ${code}`); this.name = "StyleStoreError"; }
}
export type StylesRead = { status: "ok"; styles: UserStyle[] } | { status: "newer" };
export interface StyleStoreDeps { newId(): string; nowIso(): string }

const cutName = (s: string) => Array.from(s.trim()).slice(0, STYLES.nameMax).join("");
/** A style's name as typed, trimmed and cut to `nameMax`; blank → the fallback; both blank → "My style". */
export function cleanName(name: string, fallback: string): string { return cutName(name) || cutName(fallback) || "My style"; }
const extOf = (uri: string, title: string) => ((/\.([A-Za-z0-9]+)$/.exec(uri) ?? /\.([A-Za-z0-9]+)$/.exec(title))?.[1] ?? "m4a").toLowerCase();

export function makeStyleStore(fs: FsAdapter, deps: StyleStoreDeps) {
  const dir = `${fs.documentDir}styles`;
  const file = `${dir}/styles.json`;
  const tmp = `${file}.tmp`;
  const media = `${dir}/media`;
  const musicPath = (name: string) => `${media}/${name}`;
  let chain: Promise<void> = Promise.resolve();
  let swept = false;

  /** One turn: starts when every turn asked before it has ended; a failure rejects for its caller only. Never ask for a turn inside one. */
  function inTurn<T>(work: () => Promise<T>): Promise<T> {
    const run = chain.then(work);
    chain = run.then(() => {}, () => {});
    return run;
  }
  /** The first good read of a launch removes music no style names. */
  async function sweep(styles: UserStyle[]): Promise<StylesRead> {
    if (!swept) {
      swept = true;
      const named = new Set(styles.flatMap((s) => (s.music?.kind === "own" ? [s.music.file] : [])));
      for (const name of await fs.list(media)) if (!named.has(name)) await fs.remove(musicPath(name)).catch(() => {});
    }
    return { status: "ok", styles };
  }
  async function readNow(): Promise<StylesRead> {
    if (await fs.exists(file)) {
      const read = parseStyles(await fs.readText(file));
      if (read.status === "newer") return read;
      if (read.status === "ok") return sweep(read.styles);
      await fs.move(file, `${dir}/damaged-${deps.nowIso().replace(/\D/g, "")}.json`);   // kept, never read again, never deleted
      console.warn("styles: the file could not be read; it was kept aside");
      return { status: "ok", styles: [] };
    }
    if (await fs.exists(tmp)) {
      const read = parseStyles(await fs.readText(tmp));
      if (read.status === "ok") { await fs.move(tmp, file); return sweep(read.styles); }
      return { status: "ok", styles: [] };   // a cut-off write: left where it is, the next write replaces it
    }
    return sweep([]);
  }
  async function stylesToWrite(): Promise<UserStyle[]> {
    const read = await readNow();
    if (read.status === "newer") throw new StyleStoreError("newer");
    return read.styles;
  }
  async function write(styles: UserStyle[]): Promise<void> {
    await fs.mkdir(dir);
    await fs.writeText(tmp, stylesText(styles));
    await fs.move(tmp, file);
  }

  return {
    list: (): Promise<StylesRead> => inTurn(readNow),
    add: (draft: StyleDraft, name: string, fallbackName: string): Promise<UserStyle> => inTurn(async () => {
      const styles = await stylesToWrite();
      if (styles.length >= STYLES.max) throw new StyleStoreError("full");
      const id = deps.newId();
      const { music: from, ...look } = draft;
      let music: StyleMusic | null = from?.kind === "bundled" ? { kind: "bundled", trackId: from.trackId } : null;
      let copied: string | null = null;
      if (from?.kind === "own") {
        if (!(await fs.exists(from.sourceUri))) throw new StyleStoreError("missing");
        const fileName = `${id}.${extOf(from.sourceUri, from.title)}`;
        await fs.mkdir(media);
        copied = musicPath(fileName);
        try { await fs.copy(from.sourceUri, copied); }
        catch (e) { await fs.remove(copied).catch(() => {}); throw e; }
        music = { kind: "own", file: fileName, title: from.title, durationSec: from.durationSec, beats: [...from.beats] };
      }
      const style: UserStyle = { id, name: cleanName(name, fallbackName), createdAt: deps.nowIso(), ...look, music };
      try { await write([style, ...styles]); }
      catch (e) { if (copied) await fs.remove(copied).catch(() => {}); throw e; }
      return style;
    }),
    rename: (id: string, name: string): Promise<void> => inTurn(async () => {
      const styles = await stylesToWrite();
      if (!styles.some((s) => s.id === id)) return;
      await write(styles.map((s) => (s.id === id ? { ...s, name: cleanName(name, s.name) } : s)));
    }),
    remove: (id: string): Promise<void> => inTurn(async () => {
      const styles = await stylesToWrite();
      const gone = styles.find((s) => s.id === id);
      if (!gone) return;
      const left = styles.filter((s) => s.id !== id);
      await write(left);
      const m = gone.music;
      if (m?.kind === "own" && !left.some((s) => s.music?.kind === "own" && s.music.file === m.file)) await fs.remove(musicPath(m.file)).catch(() => {});
    }),
    musicUri: (style: UserStyle): Promise<string | null> => inTurn(async () => {
      const m = style.music;
      if (m?.kind !== "own" || !isSafeName(m.file)) return null;
      return (await fs.exists(musicPath(m.file))) ? musicPath(m.file) : null;
    }),
  };
}
export type StyleStore = ReturnType<typeof makeStyleStore>;
```
